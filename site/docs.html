Quick start
# build
git clone https://github.com/1bit-MONSTER/1bit-MONSTER
cd 1bit-MONSTER && cmake -B build -DCMAKE_BUILD_TYPE=Release
cmake --build build -j

# run a model — GGUF, ONNX or native 1BP
./build/1bit zaya -m model.1bp -p "Hello world"

# serve an OpenAI-compatible API with a resident model pool
./build/1bit unified --pool

No installer yet — build from source.
