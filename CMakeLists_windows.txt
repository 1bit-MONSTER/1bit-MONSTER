# ─────────────────────────────────────────────────────────────
# CMakeLists_windows.txt — Windows MSVC build for 1bit-systems
# ─────────────────────────────────────────────────────────────
# Builds portable C++23 targets: CPU inference, Vulkan GPU,
# GGUF/1BP loaders, HTTP server, CLI tools.
#
# Usage:
#   cmake -B build -G "Visual Studio 17 2022" ^
#       -DCMAKE_TOOLCHAIN_FILE=cmake/msvc_x64.cmake ^
#       -DCMAKE_BUILD_TYPE=Release
#   cmake --build build --config Release
#
# Or simply run:   scripts\build_windows.cmd
# ─────────────────────────────────────────────────────────────

cmake_minimum_required(VERSION 3.21)
project(1bit-systems-windows LANGUAGES CXX)

# ── C++23 ──
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

if(MSVC)
    # MSVC 19.44 needs explicit /std:c++latest for C++17/20 features
    add_compile_options("/std:c++latest")
    # Force UTF-8 source/execution charset regardless of host code page —
    # narrow string literals with non-ASCII \u escapes (e.g. tokenizer.cpp's
    # SentencePiece "▁" marker) must encode as UTF-8, not codepage-1252 (#1583)
    add_compile_options("/utf-8")
    # Static MSVC runtime — no VC redist DLL dependency
    set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
else()
    set(CMAKE_CXX_STANDARD 23)
    set(CMAKE_CXX_STANDARD_REQUIRED ON)
endif()

if(NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()

# NDEBUG for Release builds
add_compile_definitions($<$<CONFIG:Release>:NDEBUG>)

# MSVC needs _USE_MATH_DEFINES for M_PI, etc.
add_compile_definitions($<$<BOOL:${MSVC}>:_USE_MATH_DEFINES>)

# ── Dependencies (vendored or FetchContent) ──
include(FetchContent)

FetchContent_Declare(cpp_httplib
    GIT_REPOSITORY https://github.com/yhirose/cpp-httplib.git
    GIT_TAG        v0.18.1)
FetchContent_Declare(nlohmann_json
    GIT_REPOSITORY https://github.com/nlohmann/json.git
    GIT_TAG        v3.11.3)
FetchContent_MakeAvailable(cpp_httplib nlohmann_json)

# ── Vulkan GPU backend (optional) ──
option(USE_VULKAN "Build ZINC Vulkan GPU backend" ON)
set(RCPP_HAVE_VULKAN FALSE)
if(USE_VULKAN)
    find_package(Vulkan QUIET)
    find_program(GLSLC_EXECUTABLE glslc)
    if(Vulkan_FOUND AND GLSLC_EXECUTABLE)
        set(RCPP_HAVE_VULKAN TRUE)
        message(STATUS "Vulkan: found at ${Vulkan_INCLUDE_DIRS}, glslc at ${GLSLC_EXECUTABLE}")
    else()
        message(WARNING "Vulkan SDK or glslc not found — building CPU-only")
    endif()
endif()

# ── Directories ──
# REPO_ROOT must point to the actual repo root (where src/, tools/, etc live)
# When this file is copied to a build directory, pass -DREPO_ROOT=... to cmake
if(NOT REPO_ROOT)
    set(REPO_ROOT ${CMAKE_CURRENT_SOURCE_DIR})
endif()
# Normalize to forward slashes — a Windows-style REPO_ROOT (e.g. from a .cmd
# script passing "C:\repo\..") makes CMake's string parser choke on sequences
# like "\1" as an invalid character escape (#1578 follow-up).
file(TO_CMAKE_PATH "${REPO_ROOT}" REPO_ROOT)
set(SRC_DIR ${REPO_ROOT}/src)
set(INCLUDE_DIR ${REPO_ROOT}/include)
set(KERNELS_DIR ${REPO_ROOT}/kernels)

# ── Include paths ──
set(BASE_INCLUDES
    ${INCLUDE_DIR}
    ${SRC_DIR}
    ${cpp_httplib_SOURCE_DIR}
    ${nlohmann_json_SOURCE_DIR}/single_include)

# ── libgguf_reader : standalone GGUF parser ──
add_library(gguf_reader STATIC
    ${SRC_DIR}/gguf_reader.cpp)
target_include_directories(gguf_reader PUBLIC ${INCLUDE_DIR})
target_compile_options(gguf_reader PRIVATE /O2)

# ── libonebp_model : mmap-backed .1bp loader (CreateFileMapping on Windows) ──
add_library(onebp_model STATIC
    ${SRC_DIR}/onebp_model.cpp)
target_include_directories(onebp_model PUBLIC ${INCLUDE_DIR})
target_compile_options(onebp_model PRIVATE /O2)

# ── libvision_encoder : ViT encoder + vision-text projector ──
# vl_processor.cpp's SSRF-guard was ported off netdb.h/arpa-inet.h to
# Winsock2 (#1587); onebp_model.cpp off mmap to CreateFileMapping — so this
# is no longer POSIX-only.
add_library(vision_encoder STATIC
    ${SRC_DIR}/vision_encoder.cpp)
target_include_directories(vision_encoder PUBLIC ${BASE_INCLUDES})
target_link_libraries(vision_encoder PUBLIC onebp_model)
target_compile_options(vision_encoder PRIVATE /O2)

# ── libwhisper (CPU) ──
add_library(whisper STATIC
    ${SRC_DIR}/whisper.cpp)
target_include_directories(whisper PUBLIC ${INCLUDE_DIR})
target_compile_options(whisper PRIVATE /O2)

# ── libbackend_manager (CPU + Vulkan backends, Windows portable subset) ──
# Only includes sources with no POSIX/HIP/ROCm dependencies.
# Files excluded for Windows portability (tracked in #1588):
#   backend_factory.cpp   — dlfcn.h (dynamic loading) + has_hip_gpu/has_vulkan/has_npu/has_avx512
#   backend_cpu.cpp       — sys/mman.h
#   backend_generic.cpp   — dirent.h
#   h1b_loader.cpp        — hip/hip_runtime.h
#   gguf_loader.cpp       — hip/hip_runtime.h
#   onnx_loader.cpp       — hip/hip_runtime.h
#   mamba2_kernels.cpp    — hip dependency path
set(BACKEND_MANAGER_SOURCES
    ${SRC_DIR}/backend_manager.cpp
    ${SRC_DIR}/backend_monitor.cpp
    ${SRC_DIR}/simple_tokenizer.cpp
    ${SRC_DIR}/q4nx_reader.cpp
    ${SRC_DIR}/safetensors_reader.cpp
    ${SRC_DIR}/model_router.cpp
    ${SRC_DIR}/tokenizer.cpp
    ${SRC_DIR}/kv_rotorquant.cpp
    ${SRC_DIR}/zamba2_engine.cpp
    # No POSIX deps — safe on Windows, just never wired in before (#1587)
    ${SRC_DIR}/dynamic_router.cpp
    ${SRC_DIR}/strategy_engine.cpp
    ${SRC_DIR}/agent_watchdog.cpp
    # netdb.h/arpa/inet.h ported to winsock2.h/ws2tcpip.h for the SSRF
    # host-resolution guard (#1587); rest of the file was already portable
    ${SRC_DIR}/vl_processor.cpp
    # dirent.h -> std::filesystem::directory_iterator (finishes the _WIN32
    # scaffolding this file already had half-written)
    ${SRC_DIR}/model_discovery.cpp
    # mmap -> CreateFileMapping, same pattern as q4nx_reader.cpp/onebp_model.cpp
    ${SRC_DIR}/unified_pool.cpp
    # has_hip_gpu/has_vulkan/has_npu/has_avx512 split out of backend_factory.cpp
    # (which stays excluded for its dlfcn.h/dirent.h backend-creation logic)
    ${SRC_DIR}/backend_detect_win.cpp
    # dlopen/dlsym/dlclose -> LoadLibrary/GetProcAddress/FreeLibrary,
    # dirent -> std::filesystem (#1588)
    ${SRC_DIR}/backend_plugin.cpp
    ${SRC_DIR}/backend_onnx.cpp)

add_library(backend_manager STATIC ${BACKEND_MANAGER_SOURCES})
target_include_directories(backend_manager PUBLIC
    ${INCLUDE_DIR}
    ${SRC_DIR}
    ${cpp_httplib_SOURCE_DIR})
target_compile_options(backend_manager PRIVATE /O2)
target_link_libraries(backend_manager PUBLIC gguf_reader)
# No Windows build of libzinc (the real GGUF tokenizer C ABI, built from a
# separate Zig repo) exists yet — fall back to simple_tokenizer.cpp's C++
# stubs (.htok/BPE or ASCII passthrough), same as CMakeLists.txt does
# whenever find_library(ZINC_LIB) comes up empty (issue #798).
target_compile_definitions(backend_manager PRIVATE ZINC_DISABLED=1)

# ONNX Runtime + VitisAI EP for NPU backend — MSVC/build-time ORT is optional;
# backend_onnx.cpp self-disables via __has_include(<onnxruntime_cxx_api.h>) when
# ORT is absent. Link only the base ORT lib (the VitisAI EP provider .so is a
# runtime plugin loaded by AppendExecutionProvider, not a link-time lib).
# ONNXRUNTIME_ROOT_DIR lets a build point at a specific ORT SDK location.
set(ONNXRUNTIME_ROOT_DIR "" CACHE PATH "ONNX Runtime install root")
find_package(onnxruntime QUIET HINTS "${ONNXRUNTIME_ROOT_DIR}")
if(onnxruntime_FOUND)
    target_include_directories(backend_manager PRIVATE ${ONNXRUNTIME_INCLUDE_DIRS})
    target_link_libraries(backend_manager PRIVATE onnxruntime::onnxruntime)
    message(STATUS "Windows ONNX backend: ONNX Runtime FOUND — VitisAI EP available")
else()
    message(STATUS "Windows ONNX backend: ONNX Runtime NOT found — ONNX_NPU backend skips ORT")
endif()

# OpenMP for CPU decode parallelization
find_package(OpenMP QUIET)
if(OpenMP_CXX_FOUND)
    target_link_libraries(backend_manager PUBLIC OpenMP::OpenMP_CXX)
    message(STATUS "OpenMP found — CPU decode is multi-threaded")
else()
    message(STATUS "OpenMP not found — CPU decode is single-threaded")
endif()

# ── ZINC Vulkan GPU backend ──
if(RCPP_HAVE_VULKAN)
    add_subdirectory(${REPO_ROOT}/engine/gpu/zinc_cpp zinc_cpp_build)
endif()

# ── Tools ──

# gguf_to_onebp : GGUF → 1BP converter
add_executable(gguf_to_onebp
    ${REPO_ROOT}/tools/gguf_to_onebp.cpp)
target_include_directories(gguf_to_onebp PRIVATE ${INCLUDE_DIR})
target_link_libraries(gguf_to_onebp PRIVATE gguf_reader)
target_compile_options(gguf_to_onebp PRIVATE /O2)

# unified_server : HTTP inference server
add_executable(unified_server
    ${REPO_ROOT}/tools/unified_server.cpp)
target_include_directories(unified_server PRIVATE ${BASE_INCLUDES})
target_link_libraries(unified_server PRIVATE backend_manager gguf_reader vision_encoder)
if(WIN32)
    target_link_libraries(unified_server PRIVATE ws2_32)
endif()
target_compile_options(unified_server PRIVATE /O2)

# ── Tests (key validation tests, CPU-only) ──
enable_testing()

add_executable(test_gguf_reader_dequant
    ${REPO_ROOT}/tests/test_gguf_reader_dequant.cpp)
target_include_directories(test_gguf_reader_dequant PRIVATE ${INCLUDE_DIR})
target_link_libraries(test_gguf_reader_dequant PRIVATE gguf_reader)
target_compile_options(test_gguf_reader_dequant PRIVATE /O2)

add_test(NAME gguf_reader_dequant COMMAND test_gguf_reader_dequant)

# ── CPack packaging ──
# Produces a self-contained ZIP with the binaries + DLLs
set(CPACK_PACKAGE_NAME "1bit-systems-windows")
set(CPACK_PACKAGE_VERSION "2026.07.26")
set(CPACK_PACKAGE_VENDOR "1bit.systems")
set(CPACK_PACKAGE_DESCRIPTION_SUMMARY "1bit-systems NPU+GPU+CPU inference engine — Windows build")
set(CPACK_PACKAGE_FILE_NAME "1bit-systems-${CPACK_PACKAGE_VERSION}-windows-x64")
set(CPACK_RESOURCE_FILE_LICENSE "${REPO_ROOT}/LICENSE")
set(CPACK_GENERATOR "ZIP")
set(CPACK_ARCHIVE_COMPONENT_INSTALL ON)

# Install targets
install(TARGETS unified_server gguf_to_onebp
    RUNTIME DESTINATION bin
    COMPONENT Runtime)
install(TARGETS backend_manager gguf_reader whisper
    ARCHIVE DESTINATION lib
    COMPONENT Runtime)
install(FILES ${REPO_ROOT}/LICENSE
    DESTINATION .
    COMPONENT Runtime)
install(FILES ${REPO_ROOT}/README.md
    DESTINATION .
    COMPONENT Runtime)

include(CPack)
